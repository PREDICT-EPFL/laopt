cmake_minimum_required(VERSION 3.21)

add_library(laopt-examples INTERFACE)
target_include_directories(laopt-examples INTERFACE include)
target_compile_options(laopt-examples INTERFACE ${laopt_compile_options})
target_link_options(laopt-examples INTERFACE ${laopt_link_options})
target_link_libraries(laopt-examples INTERFACE laopt::laopt)

add_subdirectory(double_integrator)
add_subdirectory(inverted_pendulum)
add_subdirectory(inverted_pendulum_simple)
add_subdirectory(chain_mass)

find_package(yaml-cpp QUIET)
if (TARGET yaml-cpp::yaml-cpp)
    add_subdirectory(fixed_wing)
else()
    message(STATUS "Skipping fixed_wing examples: yaml-cpp not found")
endif()

if (LAOPT_WITH_IPOPT AND LAOPT_WITH_PIQP)
    add_subdirectory(rocket)
else()
    message(STATUS "Skipping rocket example: LAOPT_WITH_IPOPT and LAOPT_WITH_PIQP are required")
endif()
